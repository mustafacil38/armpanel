name=WireGuard + Web Panel
icon=fa-solid fa-shield-halved
description=Web arayuzlu VPN sunucusu (Panel Port: 5000 / Root Gerekir)
default_port=5000
config_files=/etc/wireguard/wg0.conf,/opt/wireguard-ui/db/wireguard-ui.db
process_name=wireguard-ui,wireguard-go
command_start=export WG_QUICK_USERSPACE_IMPLEMENTATION=wireguard-go; wg-quick up wg0 2>/dev/null; cd /opt/wireguard-ui && ./wireguard-ui &
command_stop=wg-quick down wg0 2>/dev/null; pkill -f wireguard-ui 2>/dev/null || true
command_restart=pkill -f wireguard-ui 2>/dev/null; wg-quick down wg0 2>/dev/null; sleep 1; export WG_QUICK_USERSPACE_IMPLEMENTATION=wireguard-go; wg-quick up wg0 2>/dev/null; cd /opt/wireguard-ui && ./wireguard-ui &
install_script=apt update && apt install -y wireguard-tools wireguard-go iptables curl unzip
mkdir -p /etc/wireguard /opt/wireguard-ui
# Tun cihazi kontrolü ve olusturma (Root gerektirir)
if [ ! -c /dev/net/tun ]; then
    mkdir -p /dev/net
    mknod /dev/net/tun c 10 200 2>/dev/null || true
    chmod 600 /dev/net/tun 2>/dev/null || true
fi
# WireGuard-UI kurulumu
if [ ! -f /opt/wireguard-ui/wireguard-ui ]; then
    ARCH=$(uname -m)
    if [[ "$ARCH" == "aarch64" || "$ARCH" == "arm64" || "$ARCH" == "armv8"* ]]; then
        WUI_ARCH="arm64"
    else
        WUI_ARCH="amd64"
    fi
    WUI_VER=$(curl -s https://api.github.com/repos/ngoduykhanh/wireguard-ui/releases/latest | grep tag_name | cut -d'"' -f4 | tr -d v)
    wget -q "https://github.com/ngoduykhanh/wireguard-ui/releases/download/v${WUI_VER}/wireguard-ui-v${WUI_VER}-linux-${WUI_ARCH}.tar.gz" -O /tmp/wui.tar.gz
    tar xzf /tmp/wui.tar.gz -C /opt/wireguard-ui
    chmod +x /opt/wireguard-ui/wireguard-ui
    rm -f /tmp/wui.tar.gz
fi
# Temel WG yapilandirmasi yoksa olustur
if [ ! -f /etc/wireguard/wg0.conf ]; then
    PRIV_KEY=$(wg genkey)
    PUB_KEY=$(echo "${PRIV_KEY}" | wg pubkey)
    IFACE=$(ip route | grep default | awk '{print $5}' | head -1)
    cat > /etc/wireguard/wg0.conf << WGEOF
[Interface]
Address = 10.0.0.1/24
ListenPort = 51820
PrivateKey = ${PRIV_KEY}
PostUp = iptables -A FORWARD -i wg0 -j ACCEPT; iptables -t nat -A POSTROUTING -o ${IFACE} -j MASQUERADE
PostDown = iptables -D FORWARD -i wg0 -j ACCEPT; iptables -t nat -D POSTROUTING -o ${IFACE} -j MASQUERADE
WGEOF
fi
sysctl -w net.ipv4.ip_forward=1 2>/dev/null || true
echo "WireGuard + Web Panel kuruldu. Panel: http://localhost:5000"
uninstall_script=wg-quick down wg0 2>/dev/null; pkill -f wireguard-ui; rm -rf /opt/wireguard-ui /etc/wireguard
=== END ===

=== APP ===
name=Nextcloud
icon=fa-solid fa-cloud
description=Kisisel bulut depolama platformu (Port 8081)
default_port=8081
config_files=/var/www/nextcloud/config/config.php
process_name=php-fpm,nginx
command_start=nginx; php-fpm8.2 &
command_stop=pkill nginx; pkill php-fpm8.2
command_restart=pkill nginx; pkill php-fpm8.2; sleep 1; nginx; php-fpm8.2 &
install_script=apt update && apt install -y nginx php8.2-fpm php8.2-mysql php8.2-gd php8.2-curl php8.2-zip php8.2-xml php8.2-mbstring unzip
mkdir -p /var/www/nextcloud
wget -q "https://download.nextcloud.com/server/releases/latest.zip" -O /tmp/next.zip
unzip -q /tmp/next.zip -d /var/www/
chown -R www-data:www-data /var/www/nextcloud
# PHP Optimizations
sed -i "s/memory_limit = .*/memory_limit = 512M/" /etc/php/8.2/cli/php.ini
[ -f /etc/php/8.2/fpm/php.ini ] && sed -i "s/memory_limit = .*/memory_limit = 512M/" /etc/php/8.2/fpm/php.ini
sed -i 's/^listen = .*/listen = 127.0.0.1:9000/' /etc/php/8.2/fpm/pool.d/www.conf
# Nextcloud Robust Nginx Config with MJS support
cat > /etc/nginx/sites-available/nextcloud << 'EOF'
server {
    listen 8081;
    server_name _;
    root /var/www/nextcloud;
    include /etc/nginx/mime.types;
    types { application/javascript mjs; }
    add_header Referrer-Policy "no-referrer" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-XSS-Protection "1; mode=block" always;
    location = /robots.txt { allow all; log_not_found off; access_log off; }
    location = /.well-known/carddav { return 301 /remote.php/dav; }
    location = /.well-known/caldav  { return 301 /remote.php/dav; }
    client_max_body_size 512M;
    fastcgi_buffers 64 4K;
    location / { rewrite ^ /index.php; }
    location ~ ^\/(?:index|remote|public|cron|core\/ajax\/update|status|ocs\/v[12]|updater\/.+|ocm-provider\/.+)\.php(?:$|\/) {
        fastcgi_split_path_info ^(.+?\.php)(\/.*|)$;
        set $path_info $fastcgi_path_info;
        try_files $fastcgi_script_name =404;
        include fastcgi_params;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param PATH_INFO $path_info;
        fastcgi_param HTTPS off;
        fastcgi_param modHeadersAvailable true;
        fastcgi_pass 127.0.0.1:9000;
        fastcgi_intercept_errors on;
    }
    location ~ \.(?:css|js|mjs|woff2?|svg|gif|png|html|ttf|ico|jpg|jpeg)$ {
        try_files $uri /index.php$request_uri;
        access_log off;
    }
}
EOF
ln -sf /etc/nginx/sites-available/nextcloud /etc/nginx/sites-enabled/nextcloud
nginx -s reload || nginx
php-fpm8.2 &
# Post-install NC optimization
sleep 2
sudo -u www-data php /var/www/nextcloud/occ app:disable richdocuments 2>/dev/null || true
sudo -u www-data php /var/www/nextcloud/occ app:disable richdocumentscode 2>/dev/null || true
sudo -u www-data php /var/www/nextcloud/occ config:system:set trusted_domains 1 --value="*"
echo "Nextcloud kuruldu. Port: 8081"
uninstall_script=pkill -f nginx; pkill -f php-fpm
rm -rf /var/www/nextcloud /etc/nginx/sites-available/nextcloud
=== END ===
