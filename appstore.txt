name=WireGuard
icon=fa-solid fa-shield-halved
description=Hafif ve güvenli VPN tüneli
default_port=51820
config_files=/etc/wireguard/wg0.conf
process_name=wg-quick
command_start=wg-quick up wg0
command_stop=wg-quick down wg0
command_restart=wg-quick down wg0; sleep 2; wg-quick up wg0
install_script=apt install -y wireguard-tools
mkdir -p /etc/wireguard
wg genkey | tee /etc/wireguard/server_private.key | wg pubkey > /etc/wireguard/server_public.key
PRIV_KEY=$(cat /etc/wireguard/server_private.key)
cat > /etc/wireguard/wg0.conf << WGEOF
[Interface]
Address = 10.0.0.1/24
ListenPort = 51820
PrivateKey = ${PRIV_KEY}
SaveConfig = true
WGEOF
chmod 600 /etc/wireguard/wg0.conf /etc/wireguard/server_private.key
uninstall_script=wg-quick down wg0 2>/dev/null || true
apt remove -y wireguard-tools
rm -rf /etc/wireguard
=== END ===

name=Nextcloud
icon=fa-solid fa-cloud
description=Kişisel bulut depolama platformu
default_port=8081
config_files=/var/www/nextcloud/config/config.php,/etc/nginx/sites-available/nextcloud
process_name=php-fpm
command_start=nginx -s reload; php-fpm8.2
command_stop=pkill -f php-fpm
command_restart=pkill -f php-fpm; sleep 1; php-fpm8.2; nginx -s reload
install_script=mkdir -p /var/www/nextcloud /var/www/nextcloud/data
wget "https://download.nextcloud.com/server/releases/latest.zip" -O /tmp/nextcloud.zip

cp -r /tmp/nextcloud/* /var/www/nextcloud/
chown -R www-data:www-data /var/www/nextcloud
rm -rf /tmp/nextcloud /tmp/nextcloud.zip
cat > /etc/php/8.2/fpm/conf.d/99-nextcloud.ini << 'PHPEOF'
upload_max_filesize = 512M
post_max_size = 512M
memory_limit = 512M
max_execution_time = 3600
output_buffering = off
PHPEOF
cat > /etc/nginx/sites-available/nextcloud << 'NCEOF'
server {
    listen 8081;
    server_name _;
    root /var/www/nextcloud;
    index index.php;
    client_max_body_size 512M;
    add_header X-Content-Type-Options nosniff;
    add_header X-Frame-Options "SAMEORIGIN";
    add_header X-XSS-Protection "1; mode=block";
    add_header Referrer-Policy no-referrer;
    location = /.well-known/carddav { return 301 /remote.php/dav/; }
    location = /.well-known/caldav  { return 301 /remote.php/dav/; }
    location / { try_files $uri $uri/ /index.php?$query_string; }
    location ~ ^/(?:build|tests|config|lib|3rdparty|templates|data)/ { deny all; }
    location ~ \.php$ {
        include snippets/fastcgi-php.conf;
        fastcgi_pass 127.0.0.1:9000;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_param PATH_INFO $fastcgi_path_info;
    }
}
NCEOF
ln -sf /etc/nginx/sites-available/nextcloud /etc/nginx/sites-enabled/nextcloud
nginx -s reload
uninstall_script=rm -rf /var/www/nextcloud
rm -f /etc/nginx/sites-available/nextcloud /etc/nginx/sites-enabled/nextcloud
rm -f /etc/php/8.2/fpm/conf.d/99-nextcloud.ini
nginx -s reload
=== END ===

config_files=/etc/samba/smb.conf
process_name=smbd

command_stop=pkill -f smbd; pkill -f nmbd

install_script=apt install -y samba
mkdir -p /root/shared
chmod 755 /root/shared
cat > /etc/samba/smb.conf << 'SMBEOF'
[global]
   workgroup = WORKGROUP
   server string = ArmPanel Samba
   map to guest = bad user
   dns proxy = no

[shared]
   path = /root/shared
   browseable = yes
   writable = yes
   guest ok = yes
   read only = no
   create mask = 0755
   directory mask = 0755
SMBEOF
uninstall_script=pkill -f smbd 2>/dev/null || true; pkill -f nmbd 2>/dev/null || true
apt remove -y samba
rm -rf /root/shared
